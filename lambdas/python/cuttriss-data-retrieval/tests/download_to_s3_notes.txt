```bash

  --region "$AWS_REGION" --secret-id "$SECRET_NAME" \

  python3 -c 'import sys, json; s=sys.stdin.read();
from json import loads
try:
  d=loads(s); print(d.get("token") or d.get("PAT") or d.get("pat") or s, end="")
except:

```

# 2) List only files in a folder (like your `/folders/{id}/files`)

```bash
curl -sS "${SYNERGY_BASE_URL}/api/v1/folders/${START_FOLDER_ID}/files?retrieve_attributes=false&page=1&page_size=${PAGE_SIZE}&filter=&show_deleted_files=false" \
  "${hdr[@]}" | jq .
```

Grab just `(IDString, LatestVersion, Name)`:

```bash
curl -sS "${SYNERGY_BASE_URL}/api/v1/folders/${START_FOLDER_ID}/files?retrieve_attributes=false&page=1&page_size=${PAGE_SIZE}&filter=&show_deleted_files=false" \
  "${hdr[@]}" \
| jq -r '
    ( .Result // (.Files.Result // []) )[]
    | [.IDString, (.LatestVersion // 1), (.Name // "unnamed")]
    | @tsv'
```

# 3) Download one file (POST, empty body, octet-stream)

**Without references:**

```bash
FILE_ID="31_1"
VERSION="4"

curl -sS -X POST \
  "${SYNERGY_BASE_URL}/api/v1/files/${FILE_ID}/download?version=${VERSION}&with_references=false" \
  -H "Authorization: Bearer ${TOKEN}" \
  -H "Content-Type: application/octet-stream" \
  --data-binary "" \
  -o "./${FILE_ID}_v${VERSION}.bin"
```

**With references (zips):**

```bash
curl -sS -X POST \
  "${SYNERGY_BASE_URL}/api/v1/files/${FILE_ID}/download?version=${VERSION}&with_references=true" \
  -H "Authorization: Bearer ${TOKEN}" \
  -H "Content-Type: application/octet-stream" \
  --data-binary "" \
  -o "./${FILE_ID}_v${VERSION}.zip"
```

> If you see `400 You do not have access to this file!`, that’s the same permission blocker we hit in Python.

# 4) Batch: download every file in a folder to local, then upload to S3

Requires `jq` and AWS CLI.

```bash
mkdir -p dl

# pull files list
curl -sS "${SYNERGY_BASE_URL}/api/v1/folders/${START_FOLDER_ID}/files?retrieve_attributes=false&page=1&page_size=${PAGE_SIZE}&filter=&show_deleted_files=false" \
  "${hdr[@]}" \
| jq -r '
    ( .Result // (.Files.Result // []) )[]
    | [.IDString, (.LatestVersion // 1), (.Name // "unnamed")]
    | @tsv' \
| while IFS=$'\t' read -r IDSTR VER NAME; do
    # safe file name
    BASENAME="${NAME//\//_}"
    OUT="dl/${START_FOLDER_ID}_${IDSTR}_v${VER}_${BASENAME}"

    echo "[DL] ${IDSTR}@v${VER} -> ${OUT}"

    # download (POST, empty body)
    if ! curl -f -sS -X POST \
      "${SYNERGY_BASE_URL}/api/v1/files/${IDSTR}/download?version=${VER}&with_references=${WITH_REFERENCES}" \
      -H "Authorization: Bearer ${TOKEN}" \
      -H "Content-Type: application/octet-stream" \
      --data-binary "" \
      -o "${OUT}"; then
        echo "[ERR] download failed for ${IDSTR}@v${VER}" >&2
        continue
    fi

    # upload to S3 at: s3://bucket/prefix/<folder_id>/<filename>
    S3_KEY="${S3_PREFIX}/${START_FOLDER_ID}/${BASENAME}"
    echo "[S3] -> s3://${S3_BUCKET}/${S3_KEY}"
    aws s3 cp "${OUT}" "s3://${S3_BUCKET}/${S3_KEY}" \
      --content-type application/octet-stream \
      --profile "${AWS_PROFILE}" --region "${AWS_REGION}"
done
```

# 5) Recursive folder walk (like your RECURSE=True)

This does a BFS over `SubFolders` using `/folders/{id}/items`, and for each folder calls `/folders/{id}/files` and downloads/uploads as above.

```bash
#!/usr/bin/env bash
set -euo pipefail

: "${SYNERGY_BASE_URL:?}"; : "${START_FOLDER_ID:?}"; : "${PAGE_SIZE:=200}"
: "${WITH_REFERENCES:=false}"; : "${S3_BUCKET:?}"; : "${S3_PREFIX:=cuttriss/22153}"

hdr=(-H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json")

download_folder() {
  local FID="$1"

  echo "[FOLDER] ${FID}"

  # list files
  curl -sS "${SYNERGY_BASE_URL}/api/v1/folders/${FID}/files?retrieve_attributes=false&page=1&page_size=${PAGE_SIZE}&filter=&show_deleted_files=false" \
    "${hdr[@]}" \
  | jq -r '( .Result // (.Files.Result // []) )[]
           | [.IDString, (.LatestVersion // 1), (.Name // "unnamed")]
           | @tsv' \
  | while IFS=$'\t' read -r IDSTR VER NAME; do
      BASENAME="${NAME//\//_}"
      OUT="dl/${FID}_${IDSTR}_v${VER}_${BASENAME}"
      mkdir -p dl

      echo "  [DL] ${IDSTR}@v${VER} -> ${OUT}"
      if ! curl -f -sS -X POST \
        "${SYNERGY_BASE_URL}/api/v1/files/${IDSTR}/download?version=${VER}&with_references=${WITH_REFERENCES}" \
        -H "Authorization: Bearer ${TOKEN}" \
        -H "Content-Type: application/octet-stream" \
        --data-binary "" \
        -o "${OUT}"; then
          echo "  [ERR] ${IDSTR}@v${VER}" >&2
          continue
      fi

      S3_KEY="${S3_PREFIX}/${FID}/${BASENAME}"
      echo "  [S3] -> s3://${S3_BUCKET}/${S3_KEY}"
      aws s3 cp "${OUT}" "s3://${S3_BUCKET}/${S3_KEY}" \
        --content-type application/octet-stream \
        --profile "${AWS_PROFILE}" --region "${AWS_REGION}"
    done

  # list subfolders and recurse
  curl -sS "${SYNERGY_BASE_URL}/api/v1/folders/${FID}/items" "${hdr[@]}" \
  | jq -r '.SubFolders[]?.IDString' \
  | while read -r SFID; do
      download_folder "${SFID}"
    done
}

download_folder "${START_FOLDER_ID}"
```

## Notes / gotchas

* The **download** endpoint is **POST** with **empty body** and `Content-Type: application/octet-stream`. Using GET will 404/400.
* We’re pulling names from the `/files` endpoint. If anything shows as `unnamed`, switch to `retrieve_attributes=true` or enrich names from `/items`.
* The 400 **“You do not have access to this file!”** you saw will appear the same with curl. That’s server-side permission; the commands are correct.
